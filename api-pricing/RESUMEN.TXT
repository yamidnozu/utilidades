const rutaDb = path.join(__dirname, 'db.json');

async function leerDb() {
  try {
    const datos = await fs.readFile(rutaDb, 'utf8');
    return JSON.parse(datos);
  } catch (error) {
    console.error('Error leyendo db.json:', error);
    return { categorias: [], configuraciones: {}, customerPlans: [], paymentMethods: [] };
  }
}

async function escribirDb(datos) {
  try {
    await fs.writeFile(rutaDb, JSON.stringify(datos, null, 2));
  } catch (error) {
    console.error('Error escribiendo db.json:', error);
  }
}

async function simularErrores(req, res, next) {
  try {
    const db = await leerDb();
    const ruta = req.path;
    const configuracion = db.configuraciones[ruta] || 'success';

    if (configuracion === 'error') {
      return res.status(500).json({
        errors: [{
          type: "Error",
          code: "504",
          reason: "Algo salió mal",
          message: "No fue posible cargar la información. Intenta de nuevo",
          domain: ruta
        }]
      });
    }

    if (configuracion === 'empty' && ruta === '/api/v1/ch-ms-category-management-pricing/categories') {
      return res.status(404).json({
        errors: [{
          type: "Error",
          code: "CTB201",
          reason: "Lo sentimos",
          message: "Aun no tienes Categorias. Crea las categorias que deseas administrar.",
          domain: ruta
        }]
      });
    }

    next();
  } catch (error) {
    console.error('Error en simularErrores middleware:', error);
    res.status(500).json({
      errors: [{
        type: "Error",
        code: "500",
        reason: "Internal Server Error",
        message: "Error interno del servidor.",
        domain: req.path
      }]
    });
  }
}

// Crear Categoría
app.post('/api/v1/ch-ms-category-management-pricing/create-category', async (req, res) => {
  try {
    const db = await leerDb();
    const { name, description, optionList } = req.body.data.attributes.param.create;

    const nuevaCategoria = {
      categoryId: db.categorias.length + 1,
      categoryCode: `CAT${String(db.categorias.length + 1).padStart(3, '0')}`,
      categoryName: name,
      categoryDescription: description,
      categoryOptionsCount: optionList.length,
      categoryAssignedPlan: null,
      optionsList: optionList.map((opcion, index) => ({
        optionId: db.categorias.reduce((acc, cat) => acc + cat.optionsList.length, 0) + index + 1,
        optionCode: opcion.optionCode,
        optionName: opcion.description
      }))
    };

    db.categorias.push(nuevaCategoria);
    await escribirDb(db);

    res.json({
      meta: { code: "200", message: "Transacción Exitosa" },
      transactionRequest: { request: req.body },
      transactionResponse: {
        response: {
          type: "PRICINGCREATECATEGORYRS",
          id: req.body.data.id,
          attributes: {
            timestamp: new Date().toISOString(),
            appId: "MACD",
            appModule: "Pricing - Categories",
            result: nuevaCategoria
          }
        }
      }
    });
  } catch (error) {
    console.error('Error en create-category:', error);
    res.status(500).json({
      errors: [{
        type: "Error",
        code: "500",
        reason: "Internal Server Error",
        message: "No fue posible crear la categoría.",
        domain: "/api/v1/ch-ms-category-management-pricing/create-category"
      }]
    });
  }
});

// Listar Categorías
app.get('/api/v1/ch-ms-category-management-pricing/categories', async (req, res) => {
  try {
    const db = await leerDb();
    const { optionName, categoryName, pageSize = 10, pageNumber = 1, sortBy = 'categoryName' } = req.query;

    const pageSizeInt = parseInt(pageSize);
    const pageNumberInt = parseInt(pageNumber);

    let categoriasFiltradas = db.categorias;

    if (categoryName) {
      categoriasFiltradas = categoriasFiltradas.filter(cat =>
        cat.categoryName.toLowerCase().includes(categoryName.toLowerCase())
      );
    }

    if (optionName) {
      categoriasFiltradas = categoriasFiltradas.filter(cat =>
        cat.optionsList.some(opcion =>
          opcion.optionName.toLowerCase().includes(optionName.toLowerCase())
        )
      );
    }

    const totalRegistros = categoriasFiltradas.length;
    const indiceInicio = (pageNumberInt - 1) * pageSizeInt;
    const indiceFin = indiceInicio + pageSizeInt;

    categoriasFiltradas.sort((a, b) => a[sortBy].localeCompare(b[sortBy]));

    const categoriasPaginadas = categoriasFiltradas.slice(indiceInicio, indiceFin);

    if (categoriasPaginadas.length === 0) {
      return res.status(404).json({
        errors: [{
          type: "Error",
          code: "CTB202",
          reason: "Lo sentimos",
          message: "Tu búsqueda no obtuvo resultados. Prueba con otra información.",
          domain: "/api/v1/ch-ms-category-management-pricing/categories"
        }]
      });
    }

    res.json({
      meta: { code: "200", message: "Transacción Exitosa" },
      transactionRequest: { request: req.query },
      transactionResponse: {
        response: {
          type: "PRICINGRETRIEVECATEGORIESRS",
          id: null,
          attributes: {
            timestamp: new Date().toISOString(),
            appId: "MACD",
            appModule: "Pricing - Categories",
            result: {
              pagination: {
                sortBy,
                pageNumber: pageNumberInt,
                pageSize: pageSizeInt,
                totalRecords: totalRegistros
              },
              categoryList: categoriasPaginadas
            }
          }
        }
      }
    });
  } catch (error) {
    console.error('Error en listar-categorias:', error);
    res.status(500).json({
      errors: [{
        type: "Error",
        code: "500",
        reason: "Internal Server Error",
        message: "No fue posible listar las categorías.",
        domain: "/api/v1/ch-ms-category-management-pricing/categories"
      }]
    });
  }
});

app.put('/api/v1/ch-ms-relation-management-pricing/customer-plan-', async (req, res) => {

      plan.planId === currentPlanId &&
      plan.documentType === req.headers['document-type'] &&
      plan.documentNumber === req.headers['document-number']
    );

        errors: [{

app.post('/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method', async (req, res) => {
  try {
    const { documentType, documentNumber } = req.body;

    if (!documentType || !documentNumber) {
      return res.status(400).json({
        errors: [{
          type: "Error",
          code: "CTB001",
          reason: "Solicitud Incorrecta",
          message: "Faltan datos requeridos: documentType y documentNumber.",
          domain: "/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method"
        }]
      });
    }

    const db = await leerDb();
    const paymentEntry = db.paymentMethods.find(pm =>
      pm.documentType === documentType &&
      pm.documentNumber === documentNumber
    );

    if (!paymentEntry) {
      return res.status(404).json({
        errors: [{
          type: "Error",
          code: "CTB007",
          reason: "No encontrado",
          message: "El usuario no tiene un método de pago registrado.",
          domain: "/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method"
        }]
      });
    }

    res.json({
      meta: {
        code: "200",
        message: "Se ha obtenido el método de pago correctamente."
      },
      paymentMethod: {
        number: paymentEntry.paymentMethod.number,
        methodClass: paymentEntry.paymentMethod.methodClass,
        type: paymentEntry.paymentMethod.type
      }
    });
  } catch (error) {
    console.error('Error en get-payment-method:', error);
    res.status(500).json({
      errors: [{
        type: "Error",
        code: "500",
        reason: "Internal Server Error",

        domain: "/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method"

});

// Matricular o Actualizar Método de Pago de un Usuario
app.post('/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method', async (req, res) => {
  try {
    const { paymentMethod, customer } = req.body;

    if (!paymentMethod || !customer || !customer.documentType || !customer.documentNumber) {
      return res.status(400).json({
        errors: [{
          type: "Error",
          code: "CTB001",
          reason: "Solicitud Incorrecta",
          message: "Faltan datos requeridos en paymentMethod o customer.",
          domain: "/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method"
        }]
      });
    }

    const { methodClass, number, type } = paymentMethod;
    const { documentType, documentNumber } = customer;

    // Validar methodClass
    const validMethodClasses = ["CUENTA_DE_AHORRO", "CUENTA_CORRIENTE"];
    if (!validMethodClasses.includes(methodClass)) {
      return res.status(400).json({
        errors: [{
          type: "Error",
          code: "CTB008",
          reason: "Validación Fallida",
          message: "La cuenta especificada no cumple los requisitos para ser marcada como cuenta de cobro.",
          domain: "/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method"
        }]
      });
    }

    const db = await leerDb();
    const paymentEntryIndex = db.paymentMethods.findIndex(pm =>
      pm.documentType === documentType &&
      pm.documentNumber === documentNumber
    );

    if (paymentEntryIndex !== -1) {
      // Verificar si el usuario es titular de la cuenta
      const existingPaymentMethod = db.paymentMethods[paymentEntryIndex].paymentMethod;
      if (existingPaymentMethod.number !== number) {
        return res.status(400).json({
          errors: [{
            type: "Error",
            code: "CTB007",
            reason: "Lo sentimos",
            message: "El usuario no es titular de la cuenta especificada.",
            domain: "/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method"
          }]
        });
      }

      // Actualizar método de pago
      db.paymentMethods[paymentEntryIndex].paymentMethod = {
        methodClass,
        number,
        type
      };

      await escribirDb(db);

      res.json({
        matriculationSuccess: true
      });
    } else {
      // Matricular nuevo método de pago
      db.paymentMethods.push({
        documentType,
        documentNumber,
        paymentMethod: {
          methodClass,
          number,
          type
        }
      });

      await escribirDb(db);

      res.json({
        matriculationSuccess: true
      });
    }
  } catch (error) {
    console.error('Error en matriculate-payment-method:', error);
    res.status(500).json({
      errors: [{
        type: "Error",
        code: "500",
        reason: "Internal Server Error",
        message: "No fue posible matricular o actualizar el método de pago.",
        domain: "/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method"
      }]
    });
  }
});

    async function guardarConfiguraciones() {
        const configuraciones = {
            '/api/v1/ch-ms-category-management-pricing/create-category': document.getElementById('estadoCrearCategoria').value,
            '/api/v1/ch-ms-category-management-pricing/categories': document.getElementById('estadoListarCategorias').value,
            '/api/v1/ch-ms-relation-management-pricing/customer-plan': 'success',
            '/api/v1/ch-ms-relation-management-pricing/customer-plan-': 'success',
            '/api/v1/ch-ms-relation-management-pricing/customer-plan-by-document': 'success',
            '/api/v1/ch-ms-relation-management-pricing/macd/charge-percentage': 'success',
            '/api/v1/ch-ms-relation-management-pricing/macd/retrieve-customer-plan': 'success',
            '/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method': document.getElementById('estadoGetPaymentMethod').value,
            '/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method': document.getElementById('estadoMatriculatePaymentMethod').value
        };
        try {
            const response = await fetch('/api/settings', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(configuraciones)
            });
            if (response.ok) {
                toastr.success('Configuraciones guardadas exitosamente');
            } else {
                toastr.error('Error al guardar las configuraciones');
            }
        } catch (error) {
            toastr.error('Error de conexión');
        }
    }

    async function crearCategoria() {
        const nombre = document.getElementById('nombreCategoria').value;
        const descripcion = document.getElementById('descripcionCategoria').value;
        const opciones = Array.from(document.querySelectorAll('#opcionesList .flex')).map(option => ({
            optionCode: option.querySelector('.codigoOpcion').value,
            description: option.querySelector('.descripcionOpcion').value,
            inUse: true
        }));
        try {
            const response = await fetch('/api/v1/ch-ms-category-management-pricing/create-category', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    data: {
                        type: "PRICINGCREATECATEGORYRQ",
                        id: crypto.randomUUID(),
                        attributes: {
                            timestamp: new Date().toISOString(),
                            appId: "ADL",
                            appModule: "002-1",
                            param: {
                                create: {
                                    name: nombre,
                                    description: descripcion,
                                    optionList: opciones
                                }
                            }
                        }
                    }
                })
            });
            const result = await response.json();
            if (response.ok) {
                toastr.success(result.meta.message);
            } else {
                toastr.error(result.errors[0].message);
            }
        } catch (error) {
            toastr.error('Error de conexión');
        }
    }

    async function listarCategorias() {
        const nombreCategoria = document.getElementById('buscarNombreCategoria').value;
        const nombreOpcion = document.getElementById('buscarNombreOpcion').value;
        const ordenarPor = document.getElementById('ordenarPor').value;
        try {
            const response = await fetch(`/api/v1/ch-ms-category-management-pricing/categories?categoryName=${encodeURIComponent(nombreCategoria)}&optionName=${encodeURIComponent(nombreOpcion)}&sortBy=${encodeURIComponent(ordenarPor)}`);
            const result = await response.json();
            
            const listaCategorias = document.getElementById('listaCategorias');
            listaCategorias.innerHTML = '';
            
            if (result.errors) {
                toastr.warning(result.errors[0].message);
                return;
            }
            
            result.transactionResponse.response.attributes.result.categoryList.forEach(categoria => {
                const categoriaDiv = document.createElement('div');
                categoriaDiv.className = 'bg-gray-100 p-4 mb-4 rounded';
                categoriaDiv.innerHTML = `
                    <h3 class="font-bold">${categoria.categoryName}</h3>
                    <p class="text-sm text-gray-600">${categoria.categoryDescription}</p>
                    <p class="text-sm">Código: ${categoria.categoryCode}</p>
                    <p class="text-sm">Opciones: ${categoria.categoryOptionsCount}</p>
                `;
                listaCategorias.appendChild(categoriaDiv);
            });
        } catch (error) {
            toastr.error('Error al listar las categorías');
        }
    }

    async function validarNombreCategoria() {
        const nombreCategoria = document.getElementById('validarNombreCategoria').value;
        try {
            const response = await fetch(`/api/v1/ch-ms-category-management-pricing/validate-category-name?categoryName=${encodeURIComponent(nombreCategoria)}`);
            const result = await response.json();
            
            if (result.errors) {
                toastr.error(result.errors[0].message);
            } else {
                toastr.success(result.transactionResponse.response.attributes.result.description);

    async function recuperarPlanes() {
        const pageSize = document.getElementById('retrievePageSize').value || 10;
        const lastId = document.getElementById('retrieveLastId').value;
        const planId = document.getElementById('retrievePlanId').value;
        const documentType = document.getElementById('retrieveDocType').value;
        const documentNumber = document.getElementById('retrieveDocNumber').value;
        const typeService = document.getElementById('retrieveTypeService').value;

        let url = `/api/v1/ch-ms-relation-management-pricing/macd/retrieve-customer-plan?pageSize=${encodeURIComponent(pageSize)}&typeService=${encodeURIComponent(typeService)}`;
        if (lastId) url += `&lastId=${encodeURIComponent(lastId)}`;
        if (planId) url += `&planId=${encodeURIComponent(planId)}`;
        if (documentType) url += `&documentType=${encodeURIComponent(documentType)}`;
        if (documentNumber) url += `&documentNumber=${encodeURIComponent(documentNumber)}`;

        try {
            const response = await fetch(url, {
                headers: {
                    'Content-Type': 'application/json',
                    'message-id': '1234567890',
                    'session-tracker': 'abcdef123456',
                    'channel': 'web',
                    'transaction-code': '98765',
                    'request-timestamp': new Date().toISOString(),
                    'company-name': 'MiEmpresa',
                    'username': 'user123',
                    'ip': '192.168.1.1'
                }
            });

            if (!response.ok) {
                const errorData = await response.json();
                if (errorData.errors && errorData.errors.length > 0) {
                    toastr.error(errorData.errors[0].message);
                } else {
                    toastr.error('Error al recuperar los planes');
                }
                return;
            }

            const result = await response.json();

            const listaCustomerPlans = document.getElementById('listaCustomerPlans');
            listaCustomerPlans.innerHTML = '';

            if (result.data.length === 0) {
                listaCustomerPlans.innerHTML = '<p>No se encontraron planes.</p>';
                return;
            }

            result.data.forEach(plan => {
                const planDiv = document.createElement('div');
                planDiv.className = 'bg-gray-100 p-4 mb-4 rounded';
                planDiv.innerHTML = `
                    <h3 class="font-bold">Cliente: ${plan.customerInformation.customerName}</h3>
                    <p class="text-sm text-gray-600">Documento: ${plan.customerInformation.documentType} - ${plan.customerInformation.documentNumber}</p>
                    <p class="text-sm">Plan: ${plan.service[0].name}</p>
                    <p class="text-sm">Estado: ${plan.service[0].state}</p>
                    <p class="text-sm">Porcentaje de Cobro: ${plan.service[0].chargePercentage}</p>
                `;
                listaCustomerPlans.appendChild(planDiv);
            });
        } catch (error) {
            console.error('Error en recuperarPlanes:', error);
            toastr.error('Error al recuperar los planes');
        }
    }

    async function obtenerMetodoPago() {
        const documentType = document.getElementById('getPaymentDocType').value;
        const documentNumber = document.getElementById('getPaymentDocNumber').value;

        if (!documentType || !documentNumber) {
            toastr.warning('Por favor, completa ambos campos de documento.');
            return;
        }

        try {
            const response = await fetch('/api/v1/ch-ms-payment-methods-pricing/macd/get-payment-method', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    documentType,
                    documentNumber
                })
            });
            const result = await response.json();

            const mostrarMetodoPago = document.getElementById('mostrarMetodoPago');
            mostrarMetodoPago.innerHTML = '';

            if (response.ok) {
                mostrarMetodoPago.innerHTML = `
                    <div class="bg-green-100 p-4 rounded">
                        <p><strong>Número de Cuenta:</strong> ${result.paymentMethod.number}</p>
                        <p><strong>Clase:</strong> ${result.paymentMethod.methodClass}</p>
                        <p><strong>Tipo:</strong> ${result.paymentMethod.type}</p>
                    </div>
                `;
                toastr.success(result.meta.message);
            } else {
                if (result.errors && result.errors.length > 0) {
                    toastr.error(result.errors[0].message);
                } else {
                    toastr.error('Error al obtener el método de pago.');
                }
            }
        } catch (error) {
            toastr.error('Error de conexión');
        }
    }

    async function matricularMetodoPago() {
        const methodClass = document.getElementById('matPaymentClass').value;
        const number = document.getElementById('matPaymentNumber').value;
        const type = document.getElementById('matPaymentType').value;
        const documentType = document.getElementById('matCustomerDocType').value;
        const documentNumber = document.getElementById('matCustomerDocNumber').value;

        if (!methodClass || !number || !type || !documentType || !documentNumber) {
            toastr.warning('Por favor, completa todos los campos.');
            return;
        }

        try {
            const response = await fetch('/api/v1/ch-ms-payment-methods-pricing/macd/matriculate-payment-method', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    paymentMethod: {
                        methodClass,
                        number,
                        type
                    },
                    customer: {
                        documentType,
                        documentNumber
                    }
                })
            });
            const result = await response.json();

            if (response.ok) {
                toastr.success('Método de pago matriculado/actualizado exitosamente.');
                // Opcional: Limpiar los campos del formulario
                document.getElementById('matPaymentClass').value = '';
                document.getElementById('matPaymentNumber').value = '';
                document.getElementById('matPaymentType').value = '';
                document.getElementById('matCustomerDocType').value = '';
                document.getElementById('matCustomerDocNumber').value = '';
            } else {
                if (result.errors && result.errors.length > 0) {
                    toastr.error(result.errors[0].message);
                } else {
                    toastr.error('Error al matricular/actualizar el método de pago.');
                }
            }
        } catch (error) {
            toastr.error('Error de conexión');
        }
    }
